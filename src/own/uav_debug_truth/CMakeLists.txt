cmake_minimum_required(VERSION 3.16)
project(uav_debug_truth LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(gz-sim8 REQUIRED)
add_library(uav_truth_kinematics SHARED src/model_kinematics.cpp)
target_compile_features(uav_truth_kinematics PRIVATE cxx_std_17)
target_link_libraries(uav_truth_kinematics PRIVATE gz-sim8::gz-sim8)
install(TARGETS uav_truth_kinematics LIBRARY DESTINATION lib)
ament_environment_hooks(env-hooks/gz_plugin_path.dsv)
ament_package()
